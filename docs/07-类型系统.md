### 条件表达式（Condition）

用于实现：
- 有条件显示：`visibleWhen`
- 有条件编辑：`editableWhen`

```ts
type Condition =
  | { and: Condition[] }
  | { or: Condition[] }
  | { not: Condition }
  | {
    left: ConditionValueRef
    operator: ConditionOperator
    right?: ConditionValueRef
  }

type ConditionValueRef =
  | { record: string }  // 从当前记录取值：支持点路径，例如 "currency" / "meta.source.currency"
  | { global: string }  // 从全局上下文取值：支持点路径，例如 "settings.defaultCurrency"
  | { value: unknown }  // 常量值

type ConditionOperator =
  | 'eq' | 'neq'
  | 'gt' | 'gte' | 'lt' | 'lte'
  | 'in' | 'notIn'
  | 'contains' | 'startsWith' | 'endsWith'
  | 'isEmpty' | 'notEmpty'
  | 'exists' | 'notExists'
```

**编写规范**：
- `and/or/not` 可以嵌套组合；`and/or` 的数组为空时分别视为 `true/false`（与 JS every/some 行为一致）
- `record/global` 使用点路径读取值；路径不存在返回 `undefined`（可用 `exists/notExists` 判断）
- `in/notIn` 的 `right` 必须为数组常量（`{ value: [] }`）

**示例**：

```ts
editableWhen: {
  left: { record: 'currency' },
  operator: 'neq',
  right: { value: 'CNY' },
}
```

### FieldType 全部字段类型

| 类别 | 类型 | 说明 |
|------|------|------|
| 基础 | `text`, `number`, `date`, `datetime`, `boolean` | 基本类型 |
| 选择 | `select`, `multi-select`, `fk` | 下拉/外键 |
| 计算 | `formula` | 公式计算字段 |
| 数值 | `percent`, `currency` | 带格式的数字（百分比通过 decimal 控制小数位） |
| 文本 | `phone`, `email`, `url` | 带格式的文本 |
| 媒体 | `image`, `attachment` | 图片/附件 |
| 复杂 | `json` | JSON 对象 |
| 关联 | `one-to-many`, `many-to-many`, `reverse-ref` | 关系字段 |
| 特殊 | `action` | 行操作按钮列 |

## 3. service.ts — 服务类型

```typescript
// 筛选算子（10种）
type FilterOperator =
  | 'eq' | 'neq' | 'like' | 'notLike'
  | 'in' | 'notIn' | 'between' | 'notBetween'
  | 'isNull' | 'isNotNull'

interface FilterClause {
  field: string
  operator: FilterOperator
  value?: unknown       // 单值
  values?: unknown[]    // 多值（in/notIn/between/notBetween）
}

// 通用 API 响应
interface ApiResponse<T> { success, data: T, message?, errorCode?, details? }

// 列表查询参数
interface ListQueryParams {
  moduleId, filters?: FilterClause[], sort?: SortParam
  page, pageSize, cursor?, viewMode?: 'list' | 'card'
}

interface SortParam { field: string; order: 'asc' | 'desc' }

// 字段补丁参数（乐观锁）
interface PatchFieldParams { moduleId, recordId, field, value, expectedVersion, force? }
interface CreateRecordParams { moduleId, fields: Record<string, unknown> }

// 查询状态
interface QueryState { filters: FilterClause[], sort: SortParam | null, pagination: PaginationState }
interface PaginationState { page, pageSize, total, cursor? }

// 候选值查询
interface CandidateQueryParams { targetModule, keyword?, page, pageSize, filters? }
interface CandidateOption { value: string; label: string; disabled?; data? }
interface CandidateListResponse { options: CandidateOption[]; total; hasMore }

// 错误码
type ErrorCode =
  | 'VERSION_CONFLICT' | 'PERMISSION_DENIED' | 'VALIDATION_FAILED'
  | 'NOT_FOUND' | 'CIRCULAR_DEPENDENCY' | 'FORMULA_ERROR'
  | 'DYNAMIC_MAX_EXCEEDED' | 'MODULE_UNAVAILABLE' | 'SCHEMA_INVALID'
  | 'NETWORK_ERROR' | 'UNKNOWN'

interface StandardError { code: ErrorCode; message; details?; timestamp }
```
